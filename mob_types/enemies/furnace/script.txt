first_state = idling

init {
    set_var scrap 0
}

global {
    on_animation_end {
        if $wants_to_idle = true
            remove_status furnace_charge
            swallow_all
            set_var wants_to_idle false
            stop_particles
            send_message_to_links stop_aggro
            set_state idling
        end_if
    }
    on_frame_signal {
        get_event_info fs frame_signal
        if $fs = 1
            start_particles furnace_stomp 80 100
            start_particles furnace_stomp -70 -100
            stop
        end_if
        if $fs = 0
            start_particles furnace_stomp -70 100
            start_particles furnace_stomp 80 -100
            stop
        end_if
        if $fs = 2
            move_to_target focused_mob
        end_if
    }
}


script {
    idling {
        on_enter {
            stop
            set_animation idling
            get_random_int t 1 3
            set_timer $t
        }
        on_timer {
            set_state exploring
        }
    }
    exploring {
        on_enter {
            //We are going to repeatedly stop and start motion for visual effect, so we need to store the target position so he doesnt just spin in place
            get_random_float newX -1000 1000
            get_random_float newY -1000 1000
            get_mob_info myX self x
            get_mob_info myY self y
            calculate targetX $newX + $myX
            calculate targetY $newY + $myY

            follow_path_to_absolute $targetX $targetY

            set_animation walking
            get_random_int t 5 8
            set_timer $t
            set_near_reach search
        }
        on_frame_signal {
            //fs 0 and 1 handled in global
            if $fs = 2
                 follow_path_to_absolute $targetX $targetY
            end_if
        }
        on_timer {
            set_var wants_to_idle true
        }
        on_object_in_reach {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                get_mob_info target_id trigger id
                set_state chasing
            end_if
        }
        on_opponent_in_reach {
            get_mob_info target_id trigger id
            set_state chasing
        }
    }

    chasing {
        on_enter {
            send_message_to_links start_chase
            set_animation walking_eating no_restart
            set_far_reach chase
            set_near_reach charge
            focus_on_id $target_id
            move_to_target focused_mob
            start_particles furnace_incin 100 0 150
            start_chomping 15 mouth
        }
        on_frame_signal {
            //fs 0 and 1 handled in global
            if $fs = 2
                move_to_target focused_mob
            end_if
        }
        on_object_in_reach {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                set_state charging
            end_if
        }
        on_opponent_in_reach {
            set_state charging
        }
        on_focus_off_reach {
            stop_particles
            send_message_to_links stop_aggro
            set_var wants_to_idle true
        }
    }

    charging {
        on_enter {
            send_message_to_links start_charge
            stop
            set_animation walking_eating no_restart
            receive_status furnace_charge
            set_timer 6
            move_to_relative 200 0
        }
        on_frame_signal {
            //fs 0 and 1 handled in global
            if $fs = 2
                 move_to_relative 200 0
            end_if
        }
        on_hitbox_touch_a_n {
            get_mob_info mt trigger mob_type
            if $mt = mechamin
                run_next_action_as trigger
                delete
                calculate scrap $scrap + 1
                send_message_to_links consume_scrap
            end_if
        }

        //on_hitbox_touch_a_n only triggers for opponents, so we need this for scrap
        on_touch_object {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                label consume
                run_next_action_as trigger
                delete
                calculate scrap $scrap + 1
                send_message_to_links consume_scrap
            end_if
        }
        on_timer {
            set_var wants_to_idle true
        }
    }
}
