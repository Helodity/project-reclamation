first_state = idling
death_state = dying

init {
    set_var scrap 0
}

global {
    on_animation_end {
        //We only want to idle after the walking animation loops
        if $wants_to_idle = true
            remove_status furnace_charge
            swallow_all
            set_var wants_to_idle false
            stop_particles
            send_message_to_links stop_aggro
            set_state idling
        end_if
    }
    on_frame_signal {
        get_event_info fs frame_signal
        if $fs = 1
            start_particles furnace_stomp 80 100
            start_particles furnace_stomp -70 -100
            receive_status furnace_stop
            //We only want to start chasing after a step ends, to ensure acceleration remains constant
            if $wants_to_chase = true
                set_var wants_to_chase false
                set_state chasing
                end_if
            if $wants_to_charge = true
                set_var wants_to_charge false
                set_state charging
            end_if
        end_if
        if $fs = 0
            start_particles furnace_stomp -70 100
            start_particles furnace_stomp 80 -100
            receive_status furnace_stop
            if $wants_to_chase = true
                set_var wants_to_chase false
                set_state chasing
                end_if
            if $wants_to_charge = true
                set_var wants_to_charge false
                set_state charging
            end_if
        end_if
        if $fs = 2
            remove_status furnace_stop
        end_if
    }
}


script {
    idling {
        on_enter {
            stop
            set_animation idling
            get_random_int t 2 8
            set_timer $t
        }
        on_timer {
            set_state exploring
        }
    }
    exploring {
        on_enter {
            follow_path_randomly
            set_animation walking
            get_random_int t 20 120
            set_timer $t
            set_near_reach search
        }
        on_timer {
            set_var wants_to_idle true
        }
        on_object_in_reach {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                get_mob_info target_id trigger id
                set_var wants_to_chase true
            end_if
        }
        on_opponent_in_reach {
            get_mob_info target_id trigger id
            set_var wants_to_chase true
        }
        on_reach_destination {
            set_var wants_to_idle true
        }
    }

    chasing {
        on_enter {
            send_message_to_links start_chase
            set_animation walking_eating no_restart
            set_far_reach chase
            set_near_reach charge
            focus_on_id $target_id
            move_to_target focused_mob
            start_particles furnace_incin 100 0 150
            start_chomping 15 mouth
        }
        on_object_in_reach {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                set_var wants_to_charge true
            end_if
        }
        on_opponent_in_reach {
            set_var wants_to_charge true
        }
        on_focus_off_reach {
            stop_particles
            send_message_to_links stop_aggro
            set_var wants_to_idle true
        }
    }

    charging {
        on_enter {
            move_to_relative 2000 0
            send_message_to_links start_charge
            set_animation walking_eating no_restart
            receive_status furnace_charge
            set_timer 6
        }
        on_hitbox_touch_a_n {
            get_mob_info mt trigger mob_type
            if $mt = mechamin
                run_next_action_as trigger
                delete
                calculate scrap $scrap + 1
                send_message_to_links consume_scrap
            end_if
        }

        //on_hitbox_touch_a_n only triggers for opponents, so we need this for scrap
        on_touch_object {
            get_mob_info mt trigger mob_type
            if $mt = scrap
                label consume
                run_next_action_as trigger
                delete
                calculate scrap $scrap + 1
                send_message_to_links consume_scrap
            end_if
        }
        on_timer {
            set_var wants_to_idle true
        }
    }

    dying {
        on_enter {
            set_var wants_to_charge false
            set_var wants_to_chase false
            set_var wants_to_idle false
            set_animation dying
            stop
            stop_particles
            send_message_to_links stop
            start_dying
        }
        on_frame_signal {
            //Obtaining fs is set global
            if $fs = 3
                spawn explosion_one
            end_if
            if $fs = 4
                spawn explosion_two
            end_if
            if $fs = 5
                spawn explosion_three
                spawn scrap_pile
                delete
            end_if
        }
        on_animation_end {
            delete
        }
    }

}
