first_state = idling

script {
    idling {
        on_enter {
            set_animation idling
            get_random_float t 3 5
            set_timer $t
        }

        on_frame_signal {
            get_event_info fs frame_signal
            if $fs = 1
                start_particles furnace_stomp 80 100
                start_particles furnace_stomp -70 -100
                stop
            end_if
            if $fs = 0
                start_particles furnace_stomp -70 100
                start_particles furnace_stomp 80 -100
                stop
            end_if
            if $fs = 2
                 move_to_relative 100 0
            end_if
        }
        on_timer {
            set_state walking
        }
    }
    walking {
        on_enter {
            set_animation walking
            get_random_int steps 3 8
        }
        on_frame_signal {
            get_event_info fs frame_signal
            if $fs = 1
                start_particles furnace_stomp 80 100
                start_particles furnace_stomp -70 -100
                stop
            end_if
            if $fs = 0
                start_particles furnace_stomp -70 100
                start_particles furnace_stomp 80 -100
                stop
            end_if
            if $fs = 2
                 move_to_relative 100 0
            end_if
        }
        on_animation_end {
            calculate steps $steps - 1
            if $steps <= 0
                stop
                set_state idling
            end_if
        }
    }
}
