first_state = idling

script {
    idling {
        on_enter {
            set_animation idling
        }
        on_receive_message {
            get_event_info msg message

            if $msg = mechamin_activate
                set_state spawned
            end_if
        }
    }
    spawned {
        on_enter {
            start_particles mechamin_activate_friendly 0 0
            set_var spawned true
            spawn mecha
            set_timer 0.1
        }
        on_timer {
            delete
        }
    }
}
