init {
    set_var played_thump false
}

global {
    on_input_received {
        get_event_info iN input_name
        get_event_info iV input_value

        if $iN = custom_a
            if $iV = 1
                receive_status dash
            end_if
        end_if
    }

}

script {
    knocked_back {
        on_enter {
            set_var played_thump false
        }
        on_land {
            if $played_thump = false
                play_sound thump
                set_var played_thump true
            end_if
        }
    }
    ko {
        on_enter {
            set_height 20
        }
        on_leave {
            set_height 46
        }
    }
}
